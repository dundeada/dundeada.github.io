// ==================== AUDIO SYSTEM ==================== class AudioEngine { constructor() { this.audioContext = null; this.enabled = true; this.masterGain = null; this.init(); } init() { // Initialize on first user interaction document.addEventListener('click', () => this.initContext(), { once: true }); document.addEventListener('scroll', () => this.initContext(), { once: true }); } initContext() { if (this.audioContext) return; this.audioContext = new (window.AudioContext || window.webkitAudioContext)(); this.masterGain = this.audioContext.createGain(); this.masterGain.connect(this.audioContext.destination); this.masterGain.gain.value = 0.5; } playClick() { if (!this.enabled || !this.audioContext) return; const now = this.audioContext.currentTime; const osc = this.audioContext.createOscillator(); const gain = this.audioContext.createGain(); osc.connect(gain); gain.connect(this.masterGain); osc.frequency.setValueAtTime(800, now); osc.frequency.exponentialRampToValueAtTime(200, now + 0.1); gain.gain.setValueAtTime(0.3, now); gain.gain.exponentialRampToValueAtTime(0.01, now + 0.1); osc.start(now); osc.stop(now + 0.1); } playHover() { if (!this.enabled || !this.audioContext) return; const now = this.audioContext.currentTime; const osc = this.audioContext.createOscillator(); const gain = this.audioContext.createGain(); osc.connect(gain); gain.connect(this.masterGain); osc.frequency.setValueAtTime(600, now); gain.gain.setValueAtTime(0.1, now); gain.gain.exponentialRampToValueAtTime(0.01, now + 0.05); osc.start(now); osc.stop(now + 0.05); } playSubBass() { if (!this.enabled || !this.audioContext) return; const now = this.audioContext.currentTime; const osc = this.audioContext.createOscillator(); const gain = this.audioContext.createGain(); osc.connect(gain); gain.connect(this.masterGain); osc.frequency.setValueAtTime(50, now); gain.gain.setValueAtTime(0.2, now); gain.gain.exponentialRampToValueAtTime(0.01, now + 0.3); osc.start(now); osc.stop(now + 0.3); } toggle() { this.enabled = !this.enabled; if (this.masterGain) { this.masterGain.gain.value = this.enabled ? 0.5 : 0; } } } // ==================== PARTICLE SYSTEM ==================== class ParticleSystem { constructor(canvas) { this.canvas = canvas; this.ctx = canvas.getContext('2d'); this.particles = []; this.mouseX = canvas.width / 2; this.mouseY = canvas.height / 2; this.attractorX = canvas.width / 2; this.attractorY = canvas.height / 2; this.resize(); this.init(); this.loop(); window.addEventListener('resize', () => this.resize()); document.addEventListener('mousemove', (e) => this.onMouseMove(e)); } resize() { this.canvas.width = this.canvas.offsetWidth * window.devicePixelRatio; this.canvas.height = this.canvas.offsetHeight * window.devicePixelRatio; this.ctx.scale(window.devicePixelRatio, window.devicePixelRatio); } init() { this.particles = []; const count = Math.floor((this.canvas.width * this.canvas.height) / 50000); for (let i = 0; i < count; i++) { this.particles.push({ x: Math.random() * this.canvas.width / window.devicePixelRatio, y: Math.random() * this.canvas.height / window.devicePixelRatio, vx: (Math.random() - 0.5) * 2, vy: (Math.random() - 0.5) * 2, life: Math.random() * 0.5 + 0.5, }); } } onMouseMove(e) { this.mouseX = e.clientX; this.mouseY = e.clientY; this.attractorX = e.clientX; this.attractorY = e.clientY; } update() { this.particles.forEach((p) => { const dx = this.attractorX - p.x; const dy = this.attractorY - p.y; const dist = Math.sqrt(dx * dx + dy * dy); const force = Math.max(0, 100 / (dist + 1)); p.vx += (dx / (dist + 1)) * force * 0.01; p.vy += (dy / (dist + 1)) * force * 0.01; p.vx *= 0.95; p.vy *= 0.95; p.x += p.vx; p.y += p.vy; p.life -= 0.005; if (p.life <= 0) { p.x = Math.random() * this.canvas.width / window.devicePixelRatio; p.y = Math.random() * this.canvas.height / window.devicePixelRatio; p.life = 1; } }); } draw() { this.ctx.clearRect(0, 0, this.canvas.width / window.devicePixelRatio, this.canvas.height / window.devicePixelRatio); this.particles.forEach((p) => { const alpha = p.life * 0.6; const dy = p2.y - p.y; const dist = Math.sqrt(dx * dx + dy * dy); if (dist < 100) { this.ctx.strokeStyle = `rgba(0, 243, 255, ${(1 - dist / 100) * alpha * 0.3})`; this.ctx.lineWidth = 0.5; this.ctx.beginPath(); this.ctx.moveTo(p.x, p.y); this.ctx.lineTo(p2.x, p2.y); this.ctx.stroke(); } }); }); } loop() { this.update(); this.draw(); requestAnimationFrame(() => this.loop()); } } // ==================== CURSOR SYSTEM ==================== class MagneticCursor { constructor(audioEngine) { this.dot = document.querySelector('.cursor-dot');
